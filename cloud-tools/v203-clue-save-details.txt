05614                     else cached.answers.forEach{(key,answer)->if(key !in merged)merged[key]=CrosswordSession.normalize(answer)}
05615                     cached.answerOverrides.forEach{(key,answer)->merged[key]=CrosswordSession.normalize(answer)}
05616                     answerOverrideStore.putAll(cw.puzzleId,cached.answerOverrides)
05617                     val valid=merged.mapNotNull{(key,answer)->
05618                         val explanation=if(key in cached.answerOverrides) "" else cached.answerExplanations[key].orEmpty().ifBlank{currentExplanations[key].orEmpty()}
05619                         if(answer.isNotBlank()&&CompoundClues.referenceAnswerLengthMatches(cw,key,answer)) ReferenceAnswer(key.number,key.direction,answer,explanation) else null
05620                     }
05621                     if(valid.isNotEmpty()){
05622                         val next=ReferenceSet(cw.puzzleId,AnswerChecking.signature(cw),
05623                             cached.answerSourceTitle.ifBlank{current?.sourceTitle?:"פתרונות משותפים"},
05624                             cached.answerSourceUrl.ifBlank{current?.sourceUrl.orEmpty()},valid)
05625                         reference=next
05626                         referenceForPersistence=next
05627                     }
05628                 }
05629                 // File writes use fsync and can stall the UI at the exact moment a
05630                 // remote clue update arrives. Persist on the existing background
05631                 // executor, merging into the latest disk state (never an old letter
05632                 // snapshot). onPause() also saves the current board as usual.
05633                 // Persist the FULL latest canonical clue set when a new revision
05634                 // arrives. If rev N is superseded before its queued write runs,
05635                 // rev N+1 still carries all edits from N (not only its own delta).
05636                 val canonicalClues=cached.clues.toMap()
05637                 val refDelta=referenceForPersistence
05638                 val remoteOverrideKeys=cached.answerOverrides.keys.toSet()
05639                 if(canonicalClues.isNotEmpty() || refDelta!=null){
05640                     runCatching {
05641                         executor.execute {
05642                             if(canonicalIndex.revision(record.id)>cached.revision)return@execute
05643                             runCatching {
05644                                 library.mergeCanonicalPresentation(record.id,cached.gridSignature,
05645                                     canonicalClues,refDelta,remoteOverrideKeys)
05646                             }.onSuccess { persisted ->
05647                                 // Record the revision ONLY after its full local snapshot is durable.
05648                                 // If the process dies during async IO, the next open must retry
05649                                 // this revision rather than mistakenly consider it applied.
05650                                 if(persisted && canonicalIndex.revision(record.id)<=cached.revision)
05651                                     canonicalIndex.bind(record.id,issue,cached.revision,
05652                                         cached.sourceImageUrl,cached.sourceThreadUrl)
05653                             }.onFailure { error ->
05654                                 DebugLog.w("CANONICAL","live-persist-failed issue=${DebugLog.compactId(issue)} error=${error.javaClass.simpleName}")
05655                             }
05656                         }
05657                     }.onFailure { error ->
05658                         DebugLog.w("CANONICAL","live-persist-queue-failed issue=${DebugLog.compactId(issue)} error=${error.javaClass.simpleName}")
05659                     }
05660                 } else {
05661                     canonicalIndex.bind(record.id,issue,cached.revision,cached.sourceImageUrl,cached.sourceThreadUrl)
05662                 }
05663                 lastReview=null;board.refresh();rebuildCluesBelowIfNeeded()
05664                 // The source URL may arrive after first open. Retry only if this
05665                 // canonical snapshot still lacks trustworthy source-image crops.
05666                 enqueueMissingCanonicalCrops(record)
05667                 val myUid=sharing.currentUid().orEmpty()
05668                 // A single canonical snapshot may coalesce changes from different
05669                 // editors. Attribute each clue using its own committed author stamp,
05670                 // not lastEditedByName (which may refer to an answer or another clue).
05671                 val clueAlert=SharedClueUpdateAlerts.message(
05672                     previous?.let(::sharedClueAlertState),sharedClueAlertState(cached),
05673                     myUid,alreadyWatching && sharing.roomForPuzzle(record.id)!=null && !activeDuelMode
05674                 )
05675                 if(clueAlert!=null)toast(clueAlert)
05676                 else if(alreadyWatching && localRevision>0L &&
05677                     cached.lastEditedBy.isNotBlank() && cached.lastEditedBy!=myUid) {
05678                     when {
05679                         changedAnswers.size==1 -> {
05680                             val first=changedAnswers.first(); val direction=if(first.direction==Direction.HORIZONTAL)"מאוזן" else "מאונך"
05681                             toast("פתרון ${first.number} $direction עודכן")
05682                         }
05683                         changedAnswers.size>1 -> toast("עודכנו ${changedAnswers.size} פתרונות בתשבץ")
05684                     }
05685                 }
05686             }
05687         }
05688     }

05690     private fun publishCanonicalClueChanges(cw:Crossword,changes:Map<ClueEditorModel.Key,String>) {
05691         val record=runCatching{library.load(cw.puzzleId)}.getOrNull()?:return
05692         val issue=canonicalIndex.issueKeyFor(record)?:return
05693         // Publisher's structured clues are imported locally; never create or edit
05694         // production Canonical records merely because a local user edits a clue.
05695         if(issue.startsWith("INTEL:")||issue.startsWith("HIGAYON:")||issue.startsWith("ISRAEL_HAYOM:")||issue.startsWith("ACROSS14:"))return
05696         fun canonicalFailure(error:Throwable){
05697             DebugLog.w("CANONICAL", "clue-update-ui issue=${DebugLog.compactId(issue)} count=${changes.size} error=${error.javaClass.simpleName}")
05698             toast("השינוי נשמר במכשיר, אך השמירה במאגר המשותף נכשלה")
05699         }
05700         fun update(canonical:CanonicalPuzzleCloud.CachedPuzzle){
05701             canonicalCloud.updateClues(canonical.issueKey,changes,settings.displayName,
05702                 expectedGridSignature=AnswerChecking.signature(record.crossword),
05703                 discoveredNodeKey=canonical.storageNodeKey){result->
05704                 result.fold(
05705                     onSuccess={revision->canonicalIndex.setRevision(cw.puzzleId,revision)},
05706                     onFailure=::canonicalFailure
05707                 )
05708             }
05709         }
05710         if(CanonicalPuzzleCloud.isCompleteRecord(record)){
05711             canonicalForLocalEdit(issue,record){result->
05712                 result.fold(
05713                     onSuccess={cached->
05714                         if(cached.gridSignature!=AnswerChecking.signature(record.crossword)){
05715                             canonicalFailure(IllegalStateException("הרשת עודכנה על ידי מנהל. אין לשמור הגדרות של רשת ישנה"))
05716                         }else{
05717                             canonicalIndex.bind(record.id,issue,cached.revision,cached.sourceImageUrl,cached.sourceThreadUrl)
05718                             update(cached)
05719                         }
05720                     },
05721                     onFailure=::canonicalFailure
05722                 )
05723             }
05724         }
